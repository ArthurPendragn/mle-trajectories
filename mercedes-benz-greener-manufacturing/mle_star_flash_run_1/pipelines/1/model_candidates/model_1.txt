## Model name
XGBoost with Dimensionality Reduction Pipeline (PCA + ICA + TruncatedSVD)

## Example Python code
import numpy as np
import pandas as pd
from sklearn.decomposition import PCA, FastICA, TruncatedSVD
from sklearn.pipeline import FeatureUnion
import xgboost as xgb

def train_xgb_decomposition(X_train, y_train, X_val, y_val, cat_cols):
    # One-hot encode categoricals or target encode
    X_tr_enc = pd.get_dummies(X_train, columns=cat_cols, drop_first=True)
    X_val_enc = pd.get_dummies(X_val, columns=cat_cols, drop_first=True).reindex(columns=X_tr_enc.columns, fill_value=0)
    
    # Extract decomposition features
    dim_reduction = FeatureUnion([
        ('pca', PCA(n_components=10, random_state=42)),
        ('ica', FastICA(n_components=10, random_state=42, max_iter=500)),
        ('tsvd', TruncatedSVD(n_components=10, random_state=42))
    ])
    
    decomp_tr = dim_reduction.fit_transform(X_tr_enc)
    decomp_val = dim_reduction.transform(X_val_enc)
    
    X_tr_final = np.hstack([X_tr_enc.values, decomp_tr])
    X_val_final = np.hstack([X_val_enc.values, decomp_val])
    
    model = xgb.XGBRegressor(
        n_estimators=1000,
        learning_rate=0.005,
        max_depth=4,
        subsample=0.8,
        colsample_bytree=0.7,
        reg_alpha=1.0,
        reg_lambda=1.0,
        random_state=42,
        early_stopping_rounds=50
    )
    model.fit(X_tr_final, y_train, eval_set=[(X_val_final, y_val)], verbose=False)
    return model
