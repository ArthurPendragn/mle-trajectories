## Model name
Stacking Ensemble (Kernel Ridge, ElasticNet, LightGBM, and XGBoost with Ridge Meta-Learner)

## Example Python code
from sklearn.ensemble import StackingRegressor
from sklearn.linear_model import RidgeCV, ElasticNetCV
from sklearn.kernel_ridge import KernelRidge
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import RobustScaler
import lightgbm as lgb
import xgboost as xgb

def build_stacking_ensemble():
    estimators = [
        ('lgb', lgb.LGBMRegressor(n_estimators=300, learning_rate=0.02, max_depth=3, colsample_bytree=0.6, reg_alpha=1.0, random_state=42)),
        ('xgb', xgb.XGBRegressor(n_estimators=300, learning_rate=0.02, max_depth=3, colsample_bytree=0.6, reg_alpha=1.0, random_state=42)),
        ('krr', make_pipeline(RobustScaler(), KernelRidge(alpha=0.6, kernel='polynomial', degree=2, coef0=2.5))),
        ('enet', make_pipeline(RobustScaler(), ElasticNetCV(l1_ratio=[0.1, 0.5, 0.7, 0.9], cv=5, random_state=42)))
    ]
    
    stacking_reg = StackingRegressor(
        estimators=estimators,
        final_estimator=RidgeCV(alphas=[0.1, 1.0, 10.0]),
        cv=5,
        n_jobs=-1,
        passthrough=False
    )
    return stacking_reg
