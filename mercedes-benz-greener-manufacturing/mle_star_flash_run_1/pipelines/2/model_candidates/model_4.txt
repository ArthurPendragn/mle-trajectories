## Model name
Regularized Kernel Ridge / ElasticNet Stacking Pipeline

## Example Python code
import numpy as np
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import RobustScaler
from sklearn.decomposition import PCA
from sklearn.linear_model import ElasticNetCV
from sklearn.kernel_ridge import KernelRidge
from sklearn.ensemble import StackingRegressor
from sklearn.metrics import r2_score

def build_stacked_linear_kernel_model(X_train, y_train, X_val, y_val):
    # Non-tree linear and kernel models provide orthogonal variance reduction for ensembling
    base_models = [
        ('elastic', Pipeline([
            ('scaler', RobustScaler()),
            ('pca', PCA(n_components=25)),
            ('reg', ElasticNetCV(l1_ratio=[0.1, 0.5, 0.9], cv=5, random_state=42))
        ])),
        ('kernel_ridge', Pipeline([
            ('scaler', RobustScaler()),
            ('pca', PCA(n_components=20)),
            ('reg', KernelRidge(alpha=1.0, kernel='rbf', gamma=0.01))
        ]))
    ]
    
    stacking_reg = StackingRegressor(
        estimators=base_models,
        final_estimator=ElasticNetCV(cv=5, random_state=42),
        cv=5,
        n_jobs=-1
    )
    
    stacking_reg.fit(X_train, y_train)
    preds = stacking_reg.predict(X_val)
    print(f"Validation R2 Score: {r2_score(y_val, preds):.5f}")
    return stacking_reg
