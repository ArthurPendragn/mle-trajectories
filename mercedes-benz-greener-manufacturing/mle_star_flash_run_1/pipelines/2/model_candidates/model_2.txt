## Model name
XGBoost Regressor with Constrained Depth and Strong Subsampling

## Example Python code
import xgboost as xgb
from sklearn.metrics import r2_score

def train_xgboost(X_train, y_train, X_val, y_val):
    # Low max_depth and high regularization are critical to prevent overfitting on sparse binary columns
    params = {
        'objective': 'reg:squarederror',
        'eval_metric': 'rmse',
        'learning_rate': 0.008,
        'max_depth': 3,
        'subsample': 0.75,
        'colsample_bytree': 0.7,
        'alpha': 2.0,
        'lambda': 5.0,
        'random_state': 42,
        'tree_method': 'hist'
    }
    
    dtrain = xgb.DMatrix(X_train, label=y_train)
    dval = xgb.DMatrix(X_val, label=y_val)
    
    model = xgb.train(
        params,
        dtrain,
        num_boost_round=1500,
        evals=[(dtrain, 'train'), (dval, 'val')],
        early_stopping_rounds=50,
        verbose_eval=False
    )
    
    preds = model.predict(dval)
    print(f"Validation R2 Score: {r2_score(y_val, preds):.5f}")
    return model
